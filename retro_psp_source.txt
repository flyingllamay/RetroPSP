package com.retropsp.app

import android.app.Activity
import android.net.Uri
import androidx.compose.runtime.Composable
import androidx.compose.ui.platform.LocalContext
import androidx.navigation.NavType
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable
import androidx.navigation.compose.rememberNavController
import androidx.navigation.navArgument
import com.retropsp.app.monetization.AdMobHelper
import com.retropsp.app.ui.home.HomeScreen
import com.retropsp.app.ui.ingame.InGameScreen
import com.retropsp.app.ui.onboarding.OnboardingScreen
import com.retropsp.app.ui.settings.SettingsScreen

@Composable
fun RetroPspNavigation(
    hasSafPermission: Boolean,
    adMobHelper: AdMobHelper,
    onFolderSelected: (Uri) -> Unit,
    onRescanClick: () -> Unit
) {
    val navController = rememberNavController()
    val startDest = if (hasSafPermission) "home" else "onboarding"
    val activity = LocalContext.current as Activity

    NavHost(navController = navController, startDestination = startDest) {

        composable("onboarding") {
            OnboardingScreen { uri ->
                onFolderSelected(uri)
                navController.navigate("home") {
                    popUpTo("onboarding") { inclusive = true }
                }
            }
        }

        composable("home") {
            HomeScreen(
                onGameClick = { game ->
                    adMobHelper.forceShowInterstitial(activity) {
                        navController.navigate("ingame/${Uri.encode(game.uriString)}")
                    }
                },
                onSettingsClick = { navController.navigate("settings") },
                onRescanClick = {
                    adMobHelper.forceShowInterstitial(activity) {
                        onRescanClick()
                    }
                },
                onFolderSelected = onFolderSelected
            )
        }

        composable(
            route = "ingame/{uri}",
            arguments = listOf(navArgument("uri") { type = NavType.StringType })
        ) { backStack ->
            val encoded = backStack.arguments?.getString("uri") ?: ""
            InGameScreen(
                gameUriString = Uri.decode(encoded),
                adMobHelper = adMobHelper,
                onBack = {
                    adMobHelper.forceShowInterstitial(activity) {
                        navController.popBackStack()
                    }
                }
            )
        }

        composable("settings") {
            SettingsScreen(
                onBack = { navController.popBackStack() },
                onFolderSelected = { uri ->
                    onFolderSelected(uri)
                    navController.popBackStack()
                }
            )
        }
    }
}

    override fun dispatchKeyEvent(event: android.view.KeyEvent): Boolean {
        // Forward physical button presses (Gamepad, etc.) to the Native core
        val isDown = event.action == android.view.KeyEvent.ACTION_DOWN
        val isHandled = if (isDown) {
            NativeApp.keyDown(event.deviceId, event.keyCode, event.repeatCount > 0)
        } else if (event.action == android.view.KeyEvent.ACTION_UP) {
            NativeApp.keyUp(event.deviceId, event.keyCode)
        } else {
            false
        }

        // If PPSSPP handled the key (like X or Start), stop it from bubbling up to Jetpack Compose!
        if (isHandled) return true

        return super.dispatchKeyEvent(event)
    }

    override fun dispatchGenericMotionEvent(event: android.view.MotionEvent): Boolean {
        // Forward analog stick and trigger movements to the Native core
        if ((event.source and android.view.InputDevice.SOURCE_JOYSTICK) == android.view.InputDevice.SOURCE_JOYSTICK) {
            val axes = intArrayOf(
                android.view.MotionEvent.AXIS_X, android.view.MotionEvent.AXIS_Y,
                android.view.MotionEvent.AXIS_Z, android.view.MotionEvent.AXIS_RZ,
                android.view.MotionEvent.AXIS_HAT_X, android.view.MotionEvent.AXIS_HAT_Y,
                android.view.MotionEvent.AXIS_LTRIGGER, android.view.MotionEvent.AXIS_RTRIGGER
            )
            val values = FloatArray(axes.size)
            for (i in axes.indices) {
                values[i] = event.getAxisValue(axes[i])
            }
            NativeApp.joystickAxis(event.deviceId, axes, values, axes.size)
            return true
        }
        return super.dispatchGenericMotionEvent(event)
    }

package com.retropsp.app.data.saf

import android.content.Context
import android.net.Uri
import android.provider.DocumentsContract
import android.util.Log
import androidx.documentfile.provider.DocumentFile
import androidx.work.CoroutineWorker
import androidx.work.WorkerParameters
import com.retropsp.app.data.db.GameDatabase
import com.retropsp.app.data.db.GameEntity
import java.io.File

class RomScanWorker(
    appContext: Context,
    workerParams: WorkerParameters
) : CoroutineWorker(appContext, workerParams) {

    private val validExtensions = setOf("iso", "cso", "pbp", "zip")

    override suspend fun doWork(): Result {
        val folderUriString = inputData.getString("folder_uri") ?: return Result.failure()
        val folderUri = Uri.parse(folderUriString)

        val db = GameDatabase.getDatabase(applicationContext)
        val gameDao = db.gameDao()

        val games = mutableListOf<GameEntity>()
        Log.d("RomScanWorker", "Starting scan for URI: $folderUri")

        if (folderUri.scheme == "content") {
            val folder = DocumentFile.fromTreeUri(applicationContext, folderUri)
                ?: return Result.failure()
            scanContentDirectory(folder, games)
        } else {
            val path = folderUri.path ?: return Result.failure()
            val folder = File(path)
            if (!folder.exists() || !folder.isDirectory) {
                // Try fallback for content style paths if they get passed here
                return Result.failure()
            }
            scanFileDirectory(folder, games)
        }

        Log.d("RomScanWorker", "Scan complete. Found ${games.size} games.")

        gameDao.clearAll()
        gameDao.insertAll(games)

        return Result.success()
    }

    private fun scanContentDirectory(dir: DocumentFile, games: MutableList<GameEntity>) {
        dir.listFiles().forEach { file ->
            if (file.isDirectory) {
                scanContentDirectory(file, games)
            } else {
                val name = file.name ?: return@forEach
                val ext = name.substringAfterLast('.').lowercase()
                if (ext in validExtensions) {
                    val docId = DocumentsContract.getDocumentId(file.uri)
                    games.add(
                        GameEntity(
                            title = name.substringBeforeLast('.'),
                            uriString = docId
                        )
                    )
                }
            }
        }
    }

    private fun scanFileDirectory(dir: File, games: MutableList<GameEntity>) {
        dir.listFiles()?.forEach { file ->
            if (file.isDirectory) {
                scanFileDirectory(file, games)
            } else {
                val name = file.name
                val ext = name.substringAfterLast('.').lowercase()
                if (ext in validExtensions) {
                    games.add(
                        GameEntity(
                            title = name.substringBeforeLast('.'),
                            uriString = file.absolutePath
                        )
                    )
                }
            }
        }
    }
}

package com.retropsp.app.data.saf

import android.content.Context
import android.content.Intent
import android.net.Uri

class SafPermissionManager(private val context: Context) {

    fun persistPermission(uri: Uri) {
        if (uri.scheme == "content") {
            context.contentResolver.takePersistableUriPermission(
                uri,
                Intent.FLAG_GRANT_READ_URI_PERMISSION or Intent.FLAG_GRANT_WRITE_URI_PERMISSION
            )
        }
        val prefs = context.getSharedPreferences("saf_prefs", Context.MODE_PRIVATE)
        prefs.edit().putString("rom_folder_uri", uri.toString()).apply()
    }

    fun getPersistedUri(): Uri? {
        val prefs = context.getSharedPreferences("saf_prefs", Context.MODE_PRIVATE)
        val uriString = prefs.getString("rom_folder_uri", null) ?: return null
        return Uri.parse(uriString)
    }

    fun isPermissionValid(uri: Uri): Boolean {
        if (uri.scheme != "content") return true
        return context.contentResolver.persistedUriPermissions
            .any { it.uri == uri && it.isReadPermission }
    }
}

@Database(entities = [GameEntity::class], version = 1, exportSchema = false)

    companion object {
        @Volatile
        private var INSTANCE: GameDatabase? = null

        fun getDatabase(context: Context): GameDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    GameDatabase::class.java,
                    "retropsp_database"
                ).build()
                INSTANCE = instance
                instance
            }
        }
    }

@Composable
fun InGameScreen(
    gameUriString: String,
    adMobHelper: AdMobHelper,
    onBack: () -> Unit
) {
    val context = LocalContext.current
    val activity = context as Activity
    var isReadyToBoot by remember { mutableStateOf(false) }
    var bootTriggered  by remember { mutableStateOf(false) }
    var showExitDialog by remember { mutableStateOf(false) }

    val basePspDir = remember {
        File(context.getExternalFilesDir(null) ?: context.filesDir, "PSP")
    }

    val saveObserver = remember {
        SaveStateObserver(activity, adMobHelper, basePspDir)
    }

    // Intercetta il back fisico → mostra il pop-up di conferma esegui-salvataggio
    BackHandler {
        showExitDialog = true
    }

    // ── Boot Game ─────────────────────────────────────────────────────────────
    LaunchedEffect(isReadyToBoot) {
        if (!isReadyToBoot || bootTriggered) return@LaunchedEffect
        bootTriggered = true

        try {
            val treeUri = com.retropsp.app.data.saf.SafPermissionManager(context).getPersistedUri()
            if (treeUri == null) return@LaunchedEffect

            val bootPath = if (treeUri.scheme == "content") {
                val documentId = if (gameUriString.startsWith("content://")) {
                    if (gameUriString.contains("/document/")) gameUriString.substringAfter("/document/")
                    else DocumentsContract.getDocumentId(Uri.parse(gameUriString))
                } else gameUriString
                DocumentsContract.buildDocumentUriUsingTree(treeUri, documentId).toString()
            } else {
                // local file path
                gameUriString
            }

            Log.d("InGameScreen", "Booting game: $bootPath")
            NativeApp.sendMessageFromJava("shortcutParam", bootPath)
            NativeApp.sendMessageFromJava("boot", "")
            NativeApp.resume()
        } catch (e: Exception) {
            Log.e("InGameScreen", "Boot Error: ${e.message}")
        }
    }

    DisposableEffect(Unit) {
        saveObserver.start()
        onDispose { 
            saveObserver.stop()
            NativeApp.sendMessageFromJava("finish", "") 
        }
    }

    // ── Dialog di conferma uscita ──────────────────────────────────────────────
    if (showExitDialog) {
        AlertDialog(
            onDismissRequest = { showExitDialog = false },
            title = {
                Text(
                    text = "Exit Game?",
                    color = Color.White,
                    fontWeight = FontWeight.Bold,
                    fontSize = 18.sp
                )
            },
            text = {
                Text(
                    text = "Are you sure you want to return to the menu? Unsaved progress will be lost. Make sure to save your game progress inside the game before exiting!",
                    color = Color(0xFFCCCCCC),
                    fontSize = 14.sp,
                    lineHeight = 20.sp
                )
            },
            confirmButton = {
                TextButton(
                    onClick = {
                        showExitDialog = false
                        onBack()
                    }
                ) {
                    Text("Yes, Exit", color = Color(0xFFE53935), fontWeight = FontWeight.Bold)
                }
            },
            dismissButton = {
                TextButton(
                    onClick = { showExitDialog = false }
                ) {
                    Text("Cancel", color = Color(0xFF888888), fontWeight = FontWeight.Medium)
                }
            },
            containerColor = Color(0xFF151515)
        )
    }

    // ── Layout ───────────────────────────────────────────────────────────────
    Box(
        modifier = Modifier
            .fillMaxSize()

@Composable
fun InGameMenuOverlay(
    visible: Boolean,
    onDismiss: () -> Unit,
    onExit: () -> Unit
) {
    // Stato per i sotto-pannelli
    var showSaveSlots by remember { mutableStateOf(false) }
    var showLoadSlots by remember { mutableStateOf(false) }
    var isTurbo       by remember { mutableStateOf(false) }

    // Reset pannelli alla chiusura del menù
    LaunchedEffect(visible) {
        if (!visible) {
            showSaveSlots = false
            showLoadSlots = false
        }
    }

    AnimatedVisibility(
        visible = visible,
        enter = fadeIn(tween(200)),
        exit  = fadeOut(tween(200))
    ) {
        // Scrim scuro semitrasparente: click fuori chiude
        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(Color(0x88000000))
                .clickable(
                    interactionSource = remember { MutableInteractionSource() },
                    indication = null
                ) { onDismiss() },
            contentAlignment = Alignment.BottomCenter
        ) {
            AnimatedVisibility(
                visible = visible,
                enter = slideInVertically(tween(300, easing = FastOutSlowInEasing)) { it } + fadeIn(),
                exit  = slideOutVertically(tween(250)) { it } + fadeOut()
            ) {
                // Blocca i click dal propagare al scrim
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .clickable(
                            interactionSource = remember { MutableInteractionSource() },
                            indication = null
                        ) { /* intercept */ }
                ) {
                    MenuSheet(
                        isTurbo      = isTurbo,
                        showSaveSlots = showSaveSlots,
                        showLoadSlots = showLoadSlots,
                        onResume = onDismiss,
                        onSaveClick = {
                            showLoadSlots = false
                            showSaveSlots = !showSaveSlots
                        },
                        onLoadClick = {
                            showSaveSlots = false
                            showLoadSlots = !showLoadSlots
                        },
                        onSaveSlot = { slot ->
                            saveState(slot)

                        onExit = onExit
                    )
                }

        }
    }

package com.retropsp.app.monetization

import android.app.Activity
import android.content.Context
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier

import com.google.android.gms.ads.AdRequest
import com.google.android.gms.ads.AdSize
import com.google.android.gms.ads.AdView
import com.google.android.gms.ads.FullScreenContentCallback
import com.google.android.gms.ads.LoadAdError
import com.google.android.gms.ads.MobileAds
import com.google.android.gms.ads.interstitial.InterstitialAd
import com.google.android.gms.ads.interstitial.InterstitialAdLoadCallback

class AdMobHelper(private val context: Context) {

    private var interstitialAd: InterstitialAd? = null
    private var interstitialShownThisSession = false

    init {
        MobileAds.initialize(context) {}
        loadInterstitial()
    }

    private fun loadInterstitial() {
        InterstitialAd.load(
            context,
            "ca-app-pub-****************/**********", // Production Interstitial ID
            AdRequest.Builder().build(),
            object : InterstitialAdLoadCallback() {
                override fun onAdFailedToLoad(error: LoadAdError) { interstitialAd = null }
                override fun onAdLoaded(ad: InterstitialAd) { interstitialAd = ad }
            }
        )
    }

    /** Shows the interstitial once per session; calls [onDismissed] immediately if already shown. */
    fun showInterstitialOnce(activity: Activity, onDismissed: () -> Unit) {
        val ad = interstitialAd
        if (!interstitialShownThisSession && ad != null) {
            ad.fullScreenContentCallback = object : FullScreenContentCallback() {
                override fun onAdDismissedFullScreenContent() {
                    interstitialShownThisSession = true
                    onDismissed()
                    loadInterstitial() // Reload for next potential use
                }
            }
            ad.show(activity)
        } else {
            onDismissed()
        }
    }

    /** Forces an interstitial to show if loaded, calling [onDismissed] after. Reloads ad when dismissed. */
    fun forceShowInterstitial(activity: Activity, onDismissed: () -> Unit) {
        val ad = interstitialAd
        if (ad != null) {
            ad.fullScreenContentCallback = object : FullScreenContentCallback() {
                override fun onAdDismissedFullScreenContent() {
                    onDismissed()
                    loadInterstitial() // Reload for next potential use
                }
                override fun onAdFailedToShowFullScreenContent(e: com.google.android.gms.ads.AdError) {
                    onDismissed()
                }
            }
            ad.show(activity)
            interstitialAd = null // Clear reference
        } else {
            onDismissed()
        }
    }
}

package com.retropsp.app.monetization

import android.app.Activity
import android.util.Log
import kotlinx.coroutines.*
import java.io.File

/**
 * Monitors the PPSSPP save directories for changes.
 * When a save state or in-game save is created/updated, it triggers an Interstitial Ad.
 */
class SaveStateObserver(
    private val activity: Activity,
    private val adMobHelper: AdMobHelper,
    private val basePspDir: File
) {
    private var job: Job? = null
    private var lastKnownModifiedTime = 0L

    fun start() {
        if (job != null) return
        
        val saveDir = File(basePspDir, "SAVEDATA")
        val stateDir = File(basePspDir, "PPSSPP_STATE")
        
        lastKnownModifiedTime = getLatestModifiedTime(saveDir, stateDir)

        job = CoroutineScope(Dispatchers.IO).launch {
            while (isActive) {
                delay(2000) // Poll every 2 seconds
                val currentModifiedTime = getLatestModifiedTime(saveDir, stateDir)
                if (currentModifiedTime > lastKnownModifiedTime) {
                    Log.d("SaveStateObserver", "Save detected! Triggering interstitial.")
                    lastKnownModifiedTime = currentModifiedTime
                    
                    withContext(Dispatchers.Main) {
                        adMobHelper.forceShowInterstitial(activity) {}
                    }
                    
                    // Delay to prevent multiple triggers from the same save action
                    delay(5000) 
                    lastKnownModifiedTime = getLatestModifiedTime(saveDir, stateDir)
                }
            }
        }
    }

    fun stop() {
        job?.cancel()
        job = null
    }

    private fun getLatestModifiedTime(vararg dirs: File): Long {
        var latest = 0L
        for (dir in dirs) {
            if (dir.exists() && dir.isDirectory) {
                val dirMod = dir.lastModified()
                if (dirMod > latest) latest = dirMod
                
                dir.listFiles()?.forEach { child ->
                    val childMod = child.lastModified()
                    if (childMod > latest) latest = childMod
                    
                    if (child.isDirectory) {
                        child.listFiles()?.forEach { grandChild ->
                            val gcMod = grandChild.lastModified()
                            if (gcMod > latest) latest = gcMod
                        }
                    }
                }
            }
        }
        return latest
    }
}

package com.retropsp.app.monetization

import android.content.Context
import com.android.billingclient.api.*
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow

class BillingManager(private val context: Context) {

    private val _isAdFree = MutableStateFlow(false)
    val isAdFree: StateFlow<Boolean> = _isAdFree

    private val purchasesUpdatedListener = PurchasesUpdatedListener { result, purchases ->
        if (result.responseCode == BillingClient.BillingResponseCode.OK && purchases != null) {
            purchases.forEach { handlePurchase(it) }
        }
    }

    private val billingClient = BillingClient.newBuilder(context)
        .setListener(purchasesUpdatedListener)
        .enablePendingPurchases()
        .build()

    init {
        billingClient.startConnection(object : BillingClientStateListener {
            override fun onBillingSetupFinished(result: BillingResult) {
                if (result.responseCode == BillingClient.BillingResponseCode.OK) queryPurchases()
            }
            override fun onBillingServiceDisconnected() { /* retry if needed */ }
        })
    }

    private fun queryPurchases() {
        billingClient.queryPurchasesAsync(
            QueryPurchasesParams.newBuilder().setProductType(BillingClient.ProductType.INAPP).build()
        ) { result, purchases ->
            if (result.responseCode == BillingClient.BillingResponseCode.OK) {
                _isAdFree.value = purchases.any { it.products.contains("hd_pack") }
            }
        }
    }

    private fun handlePurchase(purchase: Purchase) {
        if (purchase.purchaseState == Purchase.PurchaseState.PURCHASED) {
            if (!purchase.isAcknowledged) {
                billingClient.acknowledgePurchase(
                    AcknowledgePurchaseParams.newBuilder()
                        .setPurchaseToken(purchase.purchaseToken)
                        .build()
                ) { result ->
                    if (result.responseCode == BillingClient.BillingResponseCode.OK) {
                        _isAdFree.value = true
                    }
                }
            } else {
                _isAdFree.value = true
            }
        }
    }
}
